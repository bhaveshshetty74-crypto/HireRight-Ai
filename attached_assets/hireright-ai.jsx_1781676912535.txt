// ─── Claude API helpers ───────────────────────────────────────────────────────
async function callClaude(prompt, maxTokens = 1000) {
  const res = await fetch("https://api.anthropic.com/v1/messages", {
    method: "POST",
    headers: { "Content-Type": "application/json" },
    body: JSON.stringify({ model: "claude-sonnet-4-6", max_tokens: maxTokens, messages: [{ role: "user", content: prompt }] })
  });
  const data = await res.json();
  return data.content.map(i => i.text || "").join("");
}
async function callClaudeJSON(prompt, maxTokens = 1000) {
  const text = await callClaude(prompt, maxTokens);
  return JSON.parse(text.replace(/```json|```/g, "").trim());
}

async function screenCandidate(candidate, role) {
  return callClaudeJSON(`You are an expert recruiter. Evaluate strictly.
ROLE: ${role.title} | EXP: ${role.experience} | SKILLS: ${role.skills || role.skillTags?.join(", ")}
CANDIDATE: ${candidate.name}, ${candidate.title}, ${candidate.experience}, ${candidate.location}
PROFILE: ${candidate.resume}
Return ONLY valid JSON:
{"score":<0-100>,"verdict":"Strong Fit"|"Good Fit"|"Possible Fit"|"Not a Fit","strengths":["s1","s2"],"concerns":["c1"] or [],"recommendation":"one sentence","assessmentQuestion":"one technical question"}`);
}

async function generateInterviewQuestions(candidate, role) {
  return callClaudeJSON(`Generate tailored interview questions for this candidate.
ROLE: ${role.title} | SKILLS: ${role.skills || role.skillTags?.join(", ")}
CANDIDATE: ${candidate.name}, ${candidate.title}, ${candidate.experience}
PROFILE: ${candidate.resume}
Return ONLY valid JSON:
{"technical":["q1","q2","q3","q4"],"behavioral":["q1","q2","q3"],"roleSpecific":["q1","q2","q3"],"redFlags":["thing to probe 1","thing to probe 2"]}`, 1000);
}

async function generateEmail(candidate, role, emailType) {
  const prompts = {
    invite: `Write a professional interview invite email for ${candidate.name} (${candidate.title}) for the ${role.title} role at ${role.company || "our company"}. Warm, concise, professional.`,
    reject: `Write a respectful rejection email for ${candidate.name} who applied for ${role.title} at ${role.company || "our company"}. Kind, brief, leaves door open.`,
    offer: `Write an exciting job offer email for ${candidate.name} for the ${role.title} role at ${role.company || "our company"}. Enthusiastic and professional.`,
    assessment: `Write an email to ${candidate.name} asking them to complete a technical assessment for ${role.title} at ${role.company || "our company"}. Friendly and clear.`,
  };
  const text = await callClaude(`${prompts[emailType]}\nReturn ONLY valid JSON: {"subject":"email subject","body":"full email body"}`, 800);
  return JSON.parse(text.replace(/```json|```/g, "").trim());
}

async function generateJD(role) {
  return callClaudeJSON(`Generate a job description for: Role: ${role.title}, Company: ${role.company || "a tech company"}, Experience: ${role.experience}, Skills: ${role.skills}
Return ONLY valid JSON: {"title":"title","summary":"2 sentences","responsibilities":["r1","r2","r3","r4"],"requirements":["req1","req2","req3","req4"],"niceToHave":["n1","n2"],"skillTags":["t1","t2","t3","t4","t5"]}`);
}

// ─── Utilities ────────────────────────────────────────────────────────────────
function readFileAsText(file) { return new Promise(r => { const fr = new FileReader(); fr.onload = e => r(e.target.result); fr.readAsText(file); }); }
function readFileAsDataURL(file) { return new Promise(r => { const fr = new FileReader(); fr.onload = e => r(e.target.result); fr.readAsDataURL(file); }); }

export default function HireRightAI() {

  const [roles, setRoles] = useState(SAMPLE_ROLES);
  const [activeRoleId, setActiveRoleId] = useState("r1");
  const [jds, setJds] = useState({});
  const [candidates, setCandidates] = useState(SAMPLE_CANDIDATES);
  const [screenResults, setScreenResults] = useState({});
  const [shortlisted, setShortlisted] = useState([]);
  const [toast, setToast] = useState(null);
  const [loadingStates, setLoadingStates] = useState({});

  // Modals
  const [modal, setModal] = useState(null); // { type, data }

  // Import
  const [importTab, setImportTab] = useState("paste");
  const [pasteText, setPasteText] = useState("");
  const [pasteSource, setPasteSource] = useState("LinkedIn");
  const fileInputRef = useRef();
  const csvInputRef = useRef();

  // Role form
  const [showRoleForm, setShowRoleForm] = useState(false);
  const [newRole, setNewRole] = useState({ title: "", company: "", experience: "", skills: "" });

  const activeRole = roles.find(r => r.id === activeRoleId) || roles[0];
  const roleCandidates = candidates; // In real app, filter by roleId
  const roleResults = screenResults[activeRoleId] || [];

  function setLoading(key, val) { setLoadingStates(p => ({ ...p, [key]: val })); }
  function showToast(msg, type = "success") { setToast({ msg, type }); setTimeout(() => setToast(null), 3500); }

  // ── Screening ──
  async function screenAll() {
    setLoading("screen", true);
    setScreenResults(p => ({ ...p, [activeRoleId]: [] }));
    const results = [];
    for (const c of roleCandidates) {
      try {
        const r = await screenCandidate(c, activeRole);
        results.push({ ...c, ...r, roleId: activeRoleId });
      } catch { results.push({ ...c, score: 0, verdict: "Error", strengths: [], concerns: ["Failed"], recommendation: "Manual review", assessmentQuestion: "", roleId: activeRoleId }); }
      setScreenResults(p => ({ ...p, [activeRoleId]: [...results] }));
    }
    setShortlisted(p => [...new Set([...p, ...results.filter(r => r.score >= 70).map(r => r.id)])]);
    setLoading("screen", false);
    showToast(`${results.length} candidates screened!`);
  }

  // ── Stage update ──
  function updateStage(candidateId, stage) {
    setCandidates(p => p.map(c => c.id === candidateId ? { ...c, stage } : c));
    setScreenResults(p => {
      const updated = { ...p };
      Object.keys(updated).forEach(rid => { updated[rid] = updated[rid].map(c => c.id === candidateId ? { ...c, stage } : c); });
      return updated;
    });
  }

  // ── Notes ──
  function updateNote(candidateId, note) {
    setCandidates(p => p.map(c => c.id === candidateId ? { ...c, notes: note } : c));
  }

  // ── Add role ──
  async function addRole() {
    if (!newRole.title) return;
    setLoading("addRole", true);
    const id = "r" + Date.now();
    const jd = await generateJD(newRole).catch(() => null);
    setRoles(p => [...p, { ...newRole, id }]);
    if (jd) setJds(p => ({ ...p, [id]: jd }));
    setActiveRoleId(id);
    setNewRole({ title: "", company: "", experience: "", skills: "" });
    setShowRoleForm(false);
    setLoading("addRole", false);
    showToast(`Role "${newRole.title}" added!`);
  }

  // ── PDF import ──
  async function handlePDFUpload(e) {
    const files = Array.from(e.target.files); if (!files.length) return;
    setLoading("pdf", true);
    for (const file of files) {
      try {
        const dataUrl = await readFileAsDataURL(file);
        const base64 = dataUrl.split(",")[1];
        const res = await fetch("https://api.anthropic.com/v1/messages", {
          method: "POST", headers: { "Content-Type": "application/json" },
          body: JSON.stringify({ model: "claude-sonnet-4-6", max_tokens: 600, messages: [{ role: "user", content: [{ type: "document", source: { type: "base64", media_type: "application/pdf", data: base64 } }, { type: "text", text: 'Extract resume info. Return ONLY valid JSON: {"name":"","title":"","experience":"","location":"","resume":"2-3 sentence summary"}' }] }] })
        });
        const data = await res.json();
        const parsed = JSON.parse(data.content.map(i => i.text || "").join("").replace(/```json|```/g, "").trim());
        setCandidates(p => [...p, { id: "c" + Date.now() + Math.random(), source: "PDF Upload", stage: "Applied", notes: "", ...parsed }]);
        showToast(`${parsed.name} added from PDF`);
      } catch { showToast(`Failed: ${file.name}`, "error"); }
    }
    setLoading("pdf", false); e.target.value = "";
  }

  // ── CSV import ──
  async function handleCSVUpload(e) {
    const file = e.target.files[0]; if (!file) return;
    setLoading("csv", true);
    try {
      let rows;
      if (file.name.endsWith(".xlsx") || file.name.endsWith(".xls")) {
        const wb = XLSX.read(await file.arrayBuffer(), { type: "array" });
        rows = XLSX.utils.sheet_to_json(wb.Sheets[wb.SheetNames[0]]);
      } else {
        const text = await readFileAsText(file);
        const lines = text.trim().split("\n");
        const headers = lines[0].split(",").map(h => h.trim().replace(/"/g, ""));
        rows = lines.slice(1).map(line => { const vals = line.match(/(".*?"|[^,]+)/g) || []; return Object.fromEntries(headers.map((h, i) => [h, (vals[i] || "").replace(/"/g, "").trim()])); });
      }
      const newC = rows.filter(r => r.name).map(r => ({ id: "c" + Date.now() + Math.random(), name: r.name, title: r.title || "Unknown", experience: r.experience || "Unknown", location: r.location || "Unknown", source: r.source || "CSV Import", resume: r.resume || r.skills || "", stage: "Applied", notes: "" }));
      setCandidates(p => [...p, ...newC]);
      showToast(`${newC.length} candidates imported`);
    } catch { showToast("CSV failed", "error"); }
    setLoading("csv", false); e.target.value = "";
  }

  // ── Paste import ──
  async function handlePaste() {
    if (!pasteText.trim()) return;
    setLoading("paste", true);
    try {
      const parsed = await callClaudeJSON(`Parse this ${pasteSource} profile. Return ONLY valid JSON: {"name":"","title":"","experience":"","location":"","resume":"2-3 sentence summary"}\n\nPROFILE:\n${pasteText}`);
      setCandidates(p => [...p, { id: "c" + Date.now(), source: pasteSource, stage: "Applied", notes: "", ...parsed }]);
      setPasteText(""); showToast(`${parsed.name} added from ${pasteSource}`);
    } catch { showToast("Parse failed", "error"); }
    setLoading("paste", false);
  }

  // ── Export ──

    const wb = XLSX.utils.book_new();
    XLSX.utils.book_append_sheet(wb, XLSX.utils.json_to_sheet(data), "Shortlist");
    XLSX.writeFile(wb, `${activeRole.title}_shortlist.xlsx`);
    showToast("Exported to Excel!");
  }

  // ─── Render ───────────────────────────────────────────────────────────────
  return (

"#e05c5c" : "#00c896", color: "#fff", borderRadius: 10, padding: "11px 18px", fontSize: 13, fontWeight: 600, boxShadow: "0 4px 24px #0009", maxWidth: 280 }}>{toast.type === "error" ? "⚠ " : "✓ "}{toast.msg}</div>}

            </div>

      {modal && (
        <div style={{ position: "fixed", inset: 0, background: "#000a", zIndex: 1000, display: "flex", alignItems: "center", justifyContent: "center", padding: 20 }}
          onClick={e => e.target === e.currentTarget && setModal(null)}>
          <div style={{ background: "#13151f", borderRadius: 16, padding: 24, width: "100%", maxWidth: 580, maxHeight: "85vh", overflowY: "auto", border: "1px solid #1e2235" }}>

            {/* Interview Questions Modal */}
            {modal.type === "interview" && (
              <InterviewModal candidate={modal.data.candidate} role={modal.data.role} onClose={() => setModal(null)} />
            )}

            {/* Email Modal */}
            {modal.type === "email" && (
              <EmailModal candidate={modal.data.candidate} role={modal.data.role} onClose={() => setModal(null)} />
            )}
          </div>
        </div>
      )}

      <style>{`@keyframes spin{from{transform:rotate(0deg)}to{transform:rotate(360deg)}} ::-webkit-scrollbar{width:4px;height:4px} ::-webkit-scrollbar-track{background:#0b0d14} ::-webkit-scrollbar-thumb{background:#1e2235;border-radius:4px}`}</style>
    </div>
  );
}

// ─── Interview Questions Modal ────────────────────────────────────────────────
function InterviewModal({ candidate, role, onClose }) {
  const [questions, setQuestions] = useState(null);
  const [loading, setLoading] = useState(false);

  async function generate() {
    setLoading(true);
    try {
      const q = await generateInterviewQuestions(candidate, role);
      setQuestions(q);
    } catch { alert("Generation failed. Try again."); }
    setLoading(false);
  }

  return (
    <div>
      <div style={{ display: "flex", justifyContent: "space-between", alignItems: "center", marginBottom: 18 }}>
        <div>
          <div style={{ fontSize: 16, fontWeight: 700 }}>Interview Questions</div>
          <div style={{ fontSize: 12, color: "#6b7280", marginTop: 2 }}>{candidate.name} · {candidate.title}</div>
        </div>
        <button onClick={onClose} style={{ background: "none", border: "none", color: "#6b7280", fontSize: 20, cursor: "pointer" }}>✕</button>
      </div>
      {!questions ? (
        <div style={{ textAlign: "center", padding: "30px 0" }}>
          <div style={{ fontSize: 32, marginBottom: 12 }}>🎯</div>
          <div style={{ fontSize: 13, color: "#9ca3af", marginBottom: 18 }}>Generate tailored questions based on {candidate.name}'s profile and the {role.title} role requirements.</div>
          <button onClick={generate} disabled={loading}
            style={{ background: "linear-gradient(135deg,#4f8ef7,#7c5cbf)", border: "none", borderRadius: 9, padding: "11px 24px", color: "#fff", fontWeight: 700, fontSize: 14, cursor: "pointer" }}>
            {loading ? "⏳ Generating..." : "✨ Generate Questions"}
          </button>
        </div>
      ) : (
        <div>
          {[
            { key: "technical", label: "Technical Questions", color: "#4f8ef7" },
            { key: "behavioral", label: "Behavioral Questions", color: "#00c896" },
            { key: "roleSpecific", label: "Role-Specific Questions", color: "#7c5cbf" },
            { key: "redFlags", label: "🚩 Areas to Probe", color: "#f5a623" },
          ].map(section => (
            <div key={section.key} style={{ marginBottom: 18 }}>
              <div style={{ fontSize: 11, fontWeight: 700, color: section.color, textTransform: "uppercase", letterSpacing: "0.5px", marginBottom: 10 }}>{section.label}</div>
              {questions[section.key]?.map((q, i) => (
                <div key={i} style={{ background: "#0b0d14", borderRadius: 8, padding: "10px 14px", marginBottom: 7, fontSize: 13, color: "#e8eaf0", lineHeight: 1.6, borderLeft: `3px solid ${section.color}44` }}>
                  {i + 1}. {q}
                </div>
              ))}
            </div>
          ))}
          <button onClick={() => setQuestions(null)} style={{ background: "#0b0d14", border: "1px solid #1e2235", borderRadius: 8, padding: "8px 16px", color: "#6b7280", fontSize: 12, cursor: "pointer" }}>↩ Regenerate</button>
        </div>
      )}
    </div>
  );
}

// ─── Email Modal ──────────────────────────────────────────────────────────────
function EmailModal({ candidate, role, onClose }) {
  const [emailType, setEmailType] = useState("invite");
  const [email, setEmail] = useState(null);
  const [loading, setLoading] = useState(false);
  const [copied, setCopied] = useState(false);

  async function generate() {
    setLoading(true); setEmail(null);
    try {
      const e = await generateEmail(candidate, role, emailType);
      setEmail(e);
    } catch { alert("Generation failed."); }
    setLoading(false);
  }

  function copy() {
    navigator.clipboard.writeText(`Subject: ${email.subject}\n\n${email.body}`);
    setCopied(true); setTimeout(() => setCopied(false), 2000);
  }

  return (
    <div>
      <div style={{ display: "flex", justifyContent: "space-between", alignItems: "center", marginBottom: 16 }}>
        <div>
          <div style={{ fontSize: 16, fontWeight: 700 }}>Draft Email</div>
          <div style={{ fontSize: 12, color: "#6b7280", marginTop: 2 }}>{candidate.name} · {candidate.title}</div>
        </div>
        <button onClick={onClose} style={{ background: "none", border: "none", color: "#6b7280", fontSize: 20, cursor: "pointer" }}>✕</button>
      </div>
      <div style={{ display: "grid", gridTemplateColumns: "1fr 1fr", gap: 7, marginBottom: 16 }}>
        {[
          { id: "invite", label: "📅 Interview Invite", color: "#4f8ef7" },
          { id: "assessment", label: "🧪 Send Assessment", color: "#7c5cbf" },
          { id: "offer", label: "🎉 Job Offer", color: "#00c896" },
          { id: "reject", label: "❌ Rejection", color: "#e05c5c" },
        ].map(t => (
          <button key={t.id} onClick={() => { setEmailType(t.id); setEmail(null); }}
            style={{ padding: "9px 12px", borderRadius: 9, border: `1px solid ${emailType === t.id ? t.color : "#1e2235"}`, cursor: "pointer", background: emailType === t.id ? t.color + "22" : "#0b0d14", color: emailType === t.id ? t.color : "#6b7280", fontSize: 12, fontWeight: emailType === t.id ? 700 : 400, textAlign: "left" }}>
            {t.label}
          </button>
        ))}
      </div>
      {!email ? (
        <button onClick={generate} disabled={loading}
          style={{ width: "100%", background: "linear-gradient(135deg,#4f8ef7,#7c5cbf)", border: "none", borderRadius: 9, padding: "12px 0", color: "#fff", fontWeight: 700, fontSize: 14, cursor: "pointer" }}>
          {loading ? "⏳ Drafting..." : "✨ Generate Email"}
        </button>
      ) : (
        <div>
          <div style={{ background: "#0b0d14", borderRadius: 9, padding: "10px 14px", marginBottom: 10, border: "1px solid #1e2235" }}>
            <div style={{ fontSize: 10, color: "#6b7280", textTransform: "uppercase", marginBottom: 4 }}>Subject</div>
            <div style={{ fontSize: 13, fontWeight: 600, color: "#e8eaf0" }}>{email.subject}</div>
          </div>
          <div style={{ background: "#0b0d14", borderRadius: 9, padding: "12px 14px", border: "1px solid #1e2235", fontSize: 13, color: "#b0b7c9", lineHeight: 1.7, whiteSpace: "pre-wrap", maxHeight: 260, overflowY: "auto" }}>
            {email.body}
          </div>
          <div style={{ display: "flex", gap: 8, marginTop: 12 }}>
            <button onClick={copy} style={{ flex: 1, background: copied ? "#00c89622" : "linear-gradient(135deg,#4f8ef7,#7c5cbf)", border: copied ? "1px solid #00c896" : "none", borderRadius: 8, padding: "10px 0", color: copied ? "#00c896" : "#fff", fontWeight: 700, fontSize: 13, cursor: "pointer" }}>
              {copied ? "✓ Copied!" : "📋 Copy Email"}
            </button>
            <button onClick={() => setEmail(null)} style={{ background: "#0b0d14", border: "1px solid #1e2235", borderRadius: 8, padding: "10px 14px", color: "#6b7280", fontSize: 13, cursor: "pointer" }}>↩ Redo</button>
          </div>
        </div>
      )}
    </div>
  );
}

async function generateInterviewQuestions(candidate, role) {
  const res = await fetch("https://api.anthropic.com/v1/messages", {
    method: "POST", headers: { "Content-Type": "application/json" },
    body: JSON.stringify({ model: "claude-sonnet-4-6", max_tokens: 1000, messages: [{ role: "user", content: `Generate tailored interview questions.\nROLE: ${role.title} | SKILLS: ${role.skills || ""}\nCANDIDATE: ${candidate.name}, ${candidate.title}, ${candidate.experience}\nPROFILE: ${candidate.resume}\nReturn ONLY valid JSON:\n{"technical":["q1","q2","q3","q4"],"behavioral":["q1","q2","q3"],"roleSpecific":["q1","q2","q3"],"redFlags":["probe1","probe2"]}` }] })
  });
  const data = await res.json();
  return JSON.parse(data.content.map(i => i.text || "").join("").replace(/```json|```/g, "").trim());
}

async function generateEmail(candidate, role, emailType) {
  const prompts = {
    invite: `Write a professional interview invite email for ${candidate.name} (${candidate.title}) for the ${role.title} role at ${role.company || "our company"}.`,
    reject: `Write a respectful rejection email for ${candidate.name} who applied for ${role.title} at ${role.company || "our company"}. Kind, brief, leaves door open.`,
    offer: `Write an exciting job offer email for ${candidate.name} for ${role.title} at ${role.company || "our company"}.`,
    assessment: `Write an email to ${candidate.name} asking them to complete a technical assessment for ${role.title} at ${role.company || "our company"}.`,
  };
  const res = await fetch("https://api.anthropic.com/v1/messages", {
    method: "POST", headers: { "Content-Type": "application/json" },
    body: JSON.stringify({ model: "claude-sonnet-4-6", max_tokens: 800, messages: [{ role: "user", content: `${prompts[emailType]}\nReturn ONLY valid JSON: {"subject":"email subject","body":"full email body"}` }] })
  });
  const data = await res.json();
  return JSON.parse(data.content.map(i => i.text || "").join("").replace(/```json|```/g, "").trim());
}
